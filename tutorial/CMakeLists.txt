cmake_minimum_required(VERSION 3.15)
project(AriadnePavingTutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(AriadnePaving REQUIRED)
add_executable(tutorial_paving tutorial_paving.cpp)
target_link_libraries(tutorial_paving PRIVATE AriadnePaving::ariadne-paving)
